ROUTER 1 — MIKROTIK (192.168.48.17, LAN 10.0.0.0/24)

/ip dhcp-client add interface=ether1 disabled=no add-default-route=yes
/ip address add address=10.0.0.1/24 interface=ether2
/interface wireguard add name=wg-cw5 listen-port=13231 mtu=1420
/ip address add address=10.10.10.1/24 interface=wg-cw5
:put [/interface wireguard get [find name="wg-cw5"] public-key]
/interface wireguard peers add interface=wg-cw5 public-key="KLUCZ_PUBLICZNY_ROUTERA_2" endpoint-address=192.168.48.86 endpoint-port=13231 allowed-address=10.10.10.2/32,10.0.1.0/24 persistent-keepalive=25s
/ip route add dst-address=10.0.1.0/24 gateway=wg-cw5
/ip firewall filter add chain=input action=accept protocol=udp src-address=192.168.48.86 dst-port=13231
/ip firewall filter add chain=input action=accept in-interface=wg-cw5 src-address=10.10.10.2/32
/ip firewall filter add chain=forward action=accept src-address=10.0.0.0/24 dst-address=10.0.1.0/24
/ip firewall filter add chain=forward action=accept src-address=10.0.1.0/24 dst-address=10.0.0.0/24
/ping 10.10.10.2 src-address=10.10.10.1 count=5
/ping 10.0.1.1 src-address=10.0.0.1 count=5
/interface wireguard peers print detail
